import { useState } from "react";
import { View, Text, TextInput, Button, FlatList } from "react-native";

export default function App() {
  const [list, setList] = useState([]);
  const [name, setName] = useState("");
  const [amount, setAmount] = useState("");

  const add = (type) => {
    if (!name || !amount) return;
    setList([...list, { id: Date.now().toString(), name, amount, type }]);
    setName("");
    setAmount("");
  };

  const total = list.reduce((t, i) =>
    i.type === "alacak" ? t + +i.amount : t - +i.amount, 0
  );

  return (
    <View style={{ padding: 20, marginTop: 40 }}>
      <Text style={{ fontSize: 24 }}>OMRALMA</Text>

      <TextInput placeholder="İsim" value={name} onChangeText={setName} style={{ borderWidth: 1, margin: 5 }} />
      <TextInput placeholder="Tutar" value={amount} onChangeText={setAmount} keyboardType="numeric" style={{ borderWidth: 1, margin: 5 }} />

      <Button title="Alacak" onPress={() => add("alacak")} />
      <Button title="Borç" onPress={() => add("borc")} />

      <Text>Toplam: {total} ₺</Text>

      <FlatList
        data={list}
        renderItem={({ item }) => (
          <Text>{item.name} - {item.amount} ₺ ({item.type})</Text>
        )}
      />
    </View>
  );
}
